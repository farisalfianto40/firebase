// BLOK 1: CSS & Elemen Toggle Switch 
<!DOCTYPE html>
<html>
  <head>
    <title>SWITCH CONTROL</title>
    <style>
      /* CSS Sakelar / Toggle Switch */
      .switch { position: relative; display: inline-block; width: 50px; height: 26px; }
      .switch input { opacity: 0; width: 0; height: 0; }
      .slider { position: absolute; cursor: pointer; top: 0; left: 0; right: 0; bottom: 0; background-color: #ccc; transition: .3s; border-radius: 26px; }
      .slider:before { position: absolute; content: ""; height: 18px; width: 18px; left: 4px; bottom: 4px; background-color: white; transition: .3s; border-radius: 50%; }
      input:checked + .slider { background-color: #4f46e5; }
      input:checked + .slider:before { transform: translateX(24px); }
    </style>
  </head>
  <body>
    <h1>Relay 1 Control</h1>
    
    <label class="switch">
      <input type="checkbox" id="relaySwitch">
      <span class="slider"></span>
    </label>

    <p>Status Relay 1: <span id="status_button1">OFF</span></p>
// BLOK 2: Mengambil Library Firebase SDK 
    <script type="module">
      import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-app.js";
      import { getDatabase, ref, set, onValue } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-database.js";
// BLOK 3: Konfigurasi Kredensial Firebase 
      // Ganti isi API_KEY dan databaseURL sesuai proyek Firebase kamu
      const firebaseConfig = {
        apiKey: "........",
        databaseURL: "........",
        projectId: "........"
      };

      // Menghubungkan aplikasi web ke Firebase
      const app = initializeApp(firebaseConfig);
      const db = getDatabase(app);
// BLOK 4: Logika Switch & Sinkronisasi Realtime 
      // 1. Referensi jalur data relay di database
      const relayRef = ref(db, "relay/button1");

      // 2. Ambil elemen switch & span status dari DOM
      const relaySwitch = document.getElementById("relaySwitch");
      const statusText = document.getElementById("status_button1");

      // 3. Event Listener saat Switch Digeser (Event "change")
      relaySwitch.addEventListener("change", () => {
        const valueToSend = relaySwitch.checked ? "ON" : "OFF";
        set(relayRef, valueToSend);
      });

      // 4. Listener Realtime: Update Tampilan & Posisi Switch saat Data Firebase Berubah
      onValue(relayRef, (snapshot) => {
        const status = snapshot.val();
        const currentStatus = status ? status : "OFF";
        
        // Perbarui teks status
        statusText.innerText = currentStatus;
        
        // Sinkronkan posisi sakelar agar sesuai dengan data database
        relaySwitch.checked = (currentStatus === "ON");
      });
// BLOK 5: Penutup Tag 
    </script>
  </body>
</html>
Mengapa Menggunakan Input Checkbox?

Sakelar / Toggle Switch pada web dibangun dengan elemen <input type="checkbox"> yang dihiasi CSS. Properti checked bernilai boolean (true / false) yang sangat ideal dikonversi menjadi data string "ON" atau "OFF".

Komponen Kode Fungsi & Keunggulannya
addEventListener("change", ...) Mendeteksi perubahan posisi sakelar. Lebih efisien daripada `click` karena hanya terpicu saat posisi toggle berganti.
relaySwitch.checked = (status === "ON") Sinkronisasi Dua Arah: Jika relay diubah oleh perangkat lain/ESP32, posisi fisik sakelar di web akan bergeser secara otomatis.

Kode Program ESP32 Arduino IDE (`relay_firebase.ino`)

relay_firebase.ino | Arduino IDE 2.x
ESP32 Dev Module 
#include <WiFi.h>
#include <Firebase_ESP_Client.h>

// 1. KONFIGURASI WIFI & FIREBASE
#define WIFI_SSID "........"
#define WIFI_PASSWORD "........"
#define DATABASE_URL "........"
#define DATABASE_SECRET "........"

// 2. KONFIGURASI PIN RELAY (GPIO 5)
#define RELAY_PIN 5

FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

void setup() {
  Serial.begin(115200);
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW); // Matikan relay saat awal booting

  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("Connecting to Wi-Fi");
  while (WiFi.status() != WL_CONNECTED) {
    Serial.print(".");
    delay(300);
  }
  Serial.println("\nConnected!");

  config.host = DATABASE_URL;
  config.signer.tokens.legacy_token = DATABASE_SECRET;
  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
}

void loop() {
  // Membaca status instruksi dari Firebase path: relay/button1
  if (Firebase.RTDB.getString(&fbdo, "relay/button1")) {
    String status = fbdo.stringData();
    
    if (status == "ON") {
      digitalWrite(RELAY_PIN, HIGH); // Nyalakan Relay
      Serial.println("Relay 1: ON");
    } else {
      digitalWrite(RELAY_PIN, LOW);  // Matikan Relay
      Serial.println("Relay 1: OFF");
    }
  }
  delay(500); // Cek status setiap 0.5 detik
}
