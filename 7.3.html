// BLOK 1: Structure Canvas Web & Include Chart.js 
<!DOCTYPE html>

    }

  <script src="https://cdn.jsdelivr.net/npm/chart.js"></script>
</head>
<body>

  <div class="card">
    <h2 class="card-title">Grafik Riwayat Suhu (°C)</h2>
    <div class="chart-container">
      <canvas id="sensorChart"></canvas>
    </div>
  </div>
// BLOK 2: Configuration Chart Suhu & Label Sumbu X/Y 
  <script>
    const ctx = document.getElementById('sensorChart').getContext('2d');
    const sensorChart = new Chart(ctx, {
      type: 'line',
      data: {
        labels: [],
        datasets: [
          {
            label: 'Suhu (°C)',
            data: [],
            borderColor: '#059669',
            backgroundColor: 'rgba(5, 150, 105, 0.25)',
            fill: true,
            tension: 0.3
          }
        ]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        animation: { duration: 500 },
        scales: {
          x: {
            title: {
              display: true,
              text: 'Waktu Log',
              font: { weight: 'bold' }
            }
          },
          y: {
            title: {
              display: true,
              text: 'Suhu (°C)',
              font: { weight: 'bold' }
            },
            beginAtZero: true
          }
        }
      }
    });
  </script>
// BLOK 3: Read Realtime Firebase (Membaca suhu & waktu) 
  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-app.js";
    import { getDatabase, ref, query, orderByKey, limitToLast, onValue } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-database.js";

    const firebaseConfig = {
      apiKey: "........",
      databaseURL: "........",
      projectId: "........"
    };

    const app = initializeApp(firebaseConfig);
    const db = getDatabase(app);

    // Ambil 10 data log terakhir dari node 'sensor'
    const logRef = query(ref(db, "sensor"), orderByKey(), limitToLast(10));

    onValue(logRef, (snapshot) => {
      if (!snapshot.exists()) return;

      const labels = [];
      const tempVal = [];

      snapshot.forEach((childSnapshot) => {
        const val = childSnapshot.val();
        
        // Membaca field 'waktu' untuk Sumbu X dan 'suhu' untuk Sumbu Y
        labels.push(val.waktu); 
        tempVal.push(val.suhu);
      });

      sensorChart.data.labels = labels;
      sensorChart.data.datasets[0].data = tempVal;

      sensorChart.update();
    });
  </script>
</body>
</html>
Properti / Field Penjelasan & Fungsi Utama
val.waktu Field waktu yang disimpan oleh perangkat IoT/Firebase saat data dipush. Digunakan langsung sebagai nilai label pada Sumbu X.
val.suhu Field angka penampung nilai sensor suhu (°C) yang dikirimkan perangkat ke database Realtime.
beginAtZero: true Memaksa sumbu Y untuk selalu memulai skala pengukuran dari nilai 0.
