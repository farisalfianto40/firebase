Merekam Riwayat Sensor & Menampilkan 5 Data Terakhir

Praktikum pengiriman data sensor DHT11 berkala menggunakan metode pushJSON() ke Firebase RTDB dan menampilkan 5 data riwayat terbaru secara realtime pada tabel web.

1. Skema JSON Realtime Database (`sensor`)

https://latihaniot-default-rtdb.firebaseio.com/sensor
sensor
├─ -Nxxxxx12345678 (Push Key)
├── suhu: 27.20
├── kelembapan: 75.50
└── datetime: 2026-09-05 06:23:51
└─ -Nxxxxx12345679 (Push Key Baru)
├── suhu: 27.50
├── kelembapan: 74.00
└── datetime: 2026-09-05 06:23:56

* Metode pushJSON() menghasilkan Unik Key otomatis agar data lama tidak tertimpa/overwrite.

Fungsi / Method Peran & Cara Kerja dalam Data Logger
pushJSON(&fbdo, "path", &json) Fungsi ESP32 untuk membuat ID/Key unik baru secara otomatis (berbasis timestamp Firebase) di bawah node target. Berbeda dari setFloat() yang menimpa data, pushJSON() menambah entri baru secara terus-menerus.
query(ref, orderByKey(), limitToLast(5)) Kombinasi query Web SDK untuk menyaring data dari Firebase. limitToLast(5) memastikan browser hanya mengunduh 5 log data paling baru, sehingga menghemat bandwidth internet.
snapshot.forEach(...) Iterasi/looping bawaan Firebase SDK untuk melintasi tiap anak node (child snapshot) hasil query dan memasukkannya ke dalam list array JavaScript.
ESP32 Dev Module
dht11_logger.ino 
#include <WiFi.h>
#include <Firebase_ESP_Client.h>
#include <DHT.h>
#include <time.h>

// 1. KONFIGURASI WIFI & FIREBASE
#define WIFI_SSID "........"
#define WIFI_PASSWORD "........"
#define DATABASE_URL "........"
#define DATABASE_SECRET "........"

// 2. KONFIGURASI NTP CLIENT (DATE & TIME)
const char* ntpServer = "pool.ntp.org";
const long gmtOffset_sec = 25200; // GMT+7 (WIB) = 7 * 3600
const int daylightOffset_sec = 0;

// 3. KONFIGURASI PIN SENSOR DHT11 (GPIO 4)
#define DHTPIN 4
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);
FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

String getFormattedDateTime() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    return "-";
  }
  char dateTimeBuf[25];
  strftime(dateTimeBuf, sizeof(dateTimeBuf), "%Y-%m-%d %H:%M:%S", &timeinfo);
  return String(dateTimeBuf);
}

void setup() {
  Serial.begin(115200);
  dht.begin();

  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(300);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected!");

  // Sinkronisasi Waktu dari Server NTP
  configTime(gmtOffset_sec, daylightOffset_sec, ntpServer);

  config.host = DATABASE_URL;
  config.signer.tokens.legacy_token = DATABASE_SECRET;
  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
}

void loop() {
  float t = dht.readTemperature();
  float h = dht.readHumidity();
  String dt = getFormattedDateTime();

  if (isnan(t) || isnan(h)) {
    Serial.println("Gagal membaca sensor DHT11!");
    delay(2000);
    return;
  }

  // 4. BUAT OBJEK JSON UNTUK DATA LOGGER
  FirebaseJson json;
  json.set("suhu", t);
  json.set("kelembapan", h);
  json.set("datetime", dt);

  // 5. PUSH OBJEK JSON KE NODE "sensor"
  if (Firebase.RTDB.pushJSON(&fbdo, "sensor", &json)) {
    Serial.print("Log tersimpan dengan Push ID: ");
    Serial.println(fbdo.pushName());
  } else {
    Serial.println("Gagal push log: " + fbdo.errorReason());
  }

  delay(5000); // Interval simpan log setiap 5 detik
}
